# ATT_APP telemetry. Byte layout must match
# apps/att_app/config/default_att_app_msgdefs.h; check_layout.py verifies it.
# The CCSDS header is big-endian; the payload is in the flight processor's
# byte order, little-endian for a native x86/ARM build.

TELEMETRY ATT_APP HK LITTLE_ENDIAN "ATT_APP housekeeping, 1 Hz"
  APPEND_ID_ITEM CCSDS_STREAMID 16 UINT 0x0889 "CCSDS packet identification (message ID)" BIG_ENDIAN
    FORMAT_STRING "0x%04X"
  APPEND_ITEM CCSDS_SEQUENCE 16 UINT "CCSDS sequence control" BIG_ENDIAN
    FORMAT_STRING "0x%04X"
  APPEND_ITEM CCSDS_LENGTH 16 UINT "CCSDS data length, total bytes - 7" BIG_ENDIAN
  APPEND_ITEM SECONDS 32 UINT "Time stamp, seconds" BIG_ENDIAN
  APPEND_ITEM SUBSECS 16 UINT "Time stamp, subseconds (1/65536 s)" BIG_ENDIAN
  APPEND_ITEM SPARE_FOR_64_ALIGN 32 UINT "Header padding so the payload is 8-byte aligned"
  APPEND_ITEM COMMAND_COUNTER 8 UINT "Commands accepted"
  APPEND_ITEM COMMAND_ERROR_COUNTER 8 UINT "Commands rejected (bad length, unknown code, out-of-range value)"
  APPEND_ITEM CNO_THRESHOLD 8 UINT "Current single-satellite low C/N0 threshold"
    UNITS dB-Hz dBHz
  APPEND_ITEM INPUT_STATE 8 UINT "GNSS input"
    STATE CLOSED 0 YELLOW
    STATE OPEN 1 GREEN
  APPEND_ITEM WAKEUP_COUNTER 32 UINT "50 Hz scheduler wakeups received"
  APPEND_ITEM EPOCH_COUNTER 32 UINT "GNSS epochs processed"
  APPEND_ITEM PARSE_ERROR_COUNTER 32 UINT "NMEA sentences rejected by the parser"

    UNITS dB-Hz dBHz
  # The three health bits as separate items, overlaid on HEALTH_FLAGS (byte 55).
  # Bit offsets count from the most significant bit of the byte, so bit 0x01 is 447.
  ITEM BROADBAND_DROP 445 1 UINT "Possible interference: C/N0 dropped on most satellites at once" BIG_ENDIAN
    STATE CLEAR 0 GREEN
    STATE SET 1 RED
  ITEM LOW_CNO_SAT 446 1 UINT "At least one tracked satellite below CNO_THRESHOLD" BIG_ENDIAN
    STATE CLEAR 0 GREEN
    STATE SET 1 YELLOW
  ITEM NO_FIX 447 1 UINT "Receiver reports no position fix" BIG_ENDIAN
    STATE CLEAR 0 GREEN
    STATE SET 1 RED
